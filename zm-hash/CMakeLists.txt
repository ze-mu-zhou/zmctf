cmake_minimum_required(VERSION 3.25)
project(zm_hash LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(zm-hash src/main.cpp src/md5_avx512.cpp)
target_compile_features(zm-hash PRIVATE cxx_std_26)
find_package(Threads REQUIRED)
target_link_libraries(zm-hash PRIVATE Threads::Threads)

option(ZM_HASH_ENABLE_AVX512 "Build the optional AVX-512 MD5 kernel" ON)
option(ZM_HASH_USE_LIBDIVIDE "Use libdivide for runtime mixed-radix decoding" ON)
set(ZM_HASH_AVX512_COMPILED OFF)
if (ZM_HASH_ENABLE_AVX512)
  include(CheckCXXCompilerFlag)
  if (MSVC)
    check_cxx_compiler_flag("/arch:AVX512" ZM_HASH_HAS_MSVC_AVX512)
    if (ZM_HASH_HAS_MSVC_AVX512)
      set_property(SOURCE src/md5_avx512.cpp APPEND PROPERTY COMPILE_OPTIONS /arch:AVX512)
      set_property(SOURCE src/md5_avx512.cpp APPEND PROPERTY COMPILE_DEFINITIONS ZM_HASH_AVX512_COMPILED=1)
      set(ZM_HASH_AVX512_COMPILED ON)
    endif()
  elseif (CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    check_cxx_compiler_flag("-mavx512f" ZM_HASH_HAS_AVX512F)
    if (ZM_HASH_HAS_AVX512F)
      set_property(SOURCE src/md5_avx512.cpp APPEND PROPERTY COMPILE_OPTIONS -mavx512f)
      set_property(SOURCE src/md5_avx512.cpp APPEND PROPERTY COMPILE_DEFINITIONS ZM_HASH_AVX512_COMPILED=1)
      set(ZM_HASH_AVX512_COMPILED ON)
    endif()
  endif()
endif()
message(STATUS "AVX-512 kernel: ${ZM_HASH_AVX512_COMPILED}")

if (ZM_HASH_USE_LIBDIVIDE)
  target_include_directories(zm-hash PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/libdivide)
  target_compile_definitions(zm-hash PRIVATE ZM_HASH_USE_LIBDIVIDE=1)
endif()

if (MSVC)
  target_compile_options(zm-hash PRIVATE /O2 /W4 /permissive-)
else()
  target_compile_options(zm-hash PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()

set(ZM_HASH_EXTRA_CXX_FLAGS "" CACHE STRING "Extra compiler flags appended after the default optimization flags")
if (ZM_HASH_EXTRA_CXX_FLAGS)
  separate_arguments(ZM_HASH_EXTRA_CXX_FLAGS_LIST NATIVE_COMMAND "${ZM_HASH_EXTRA_CXX_FLAGS}")
  target_compile_options(zm-hash PRIVATE ${ZM_HASH_EXTRA_CXX_FLAGS_LIST})
endif()

include(CTest)
if (BUILD_TESTING)
  add_test(NAME md5_empty COMMAND zm-hash --mode hash --text "")
endif()
